# Shell execution corpus: each entry is a script, the stdout it must print and
# its exit status. check_bash.py runs every entry through bash too, so this file
# records what bash does; the host tests check that BMDS does the same.
#
#   === name
#   <script lines>
#   --- out
#   <expected stdout lines; none for no output. Blank lines count, except
#    trailing ones (the separator before the next entry)>
#   --- status N            (optional; default 0)
#
# Only stdout and the status are compared (error messages go to stderr and are
# worded per shell). Scripts use only commands both shells have.

=== echo and quoting
echo hello   world
echo 'single  $x' "double  quoted"
echo a\ b "c\"d" 'e'"f"g
--- out
hello world
single  $x double  quoted
a b c"d efg
